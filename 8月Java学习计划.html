import java.util.Scanner;

public class IfDemo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("输入分数：");
        int score = sc.nextInt();

        // if-else if：从上往下，命中一个就停
        if (score >= 90) {
            System.out.println("优秀");
        } else if (score >= 60) {
            System.out.println("及格");
        } else {
            System.out.println("不及格");
        }

        // switch：根据一个值"跳转"（箭头语法更简洁）
        System.out.print("输入菜单项 1-3：");

        switch (opt) {
            case 1 -> System.out.println("打开文件");
            case 2 -> System.out.println("保存文件");
            case 3 -> System.out.println("退出程序");
            default -> System.out.println("无效选项");
        }
    }
}

// 九九乘法表 —— 嵌套循环经典
public class LoopDemo {
    public static void main(String[] args) {
        for (int i = 1; i <= 9; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print(j + "x" + i + "=" + (i * j) + "\t");
            }
            System.out.println();   // 每行结束换行
        }

        // 猜数字（随机数）
        int target = (int) (Math.random() * 100) + 1;   // 1-100 随机
        java.util.Scanner sc = new java.util.Scanner(System.in);
        int guess;
        do {
            System.out.print("猜一个数：");
            guess = sc.nextInt();
            if (guess > target) System.out.println("大了");
            else if (guess < target) System.out.println("小了");
        } while (guess != target);
        System.out.println("猜对了！");
    }
}

Day 4（8.8）数组 批量数据的第一个容器
  • 数组三件套：声明 int[] arr = new int[5]、遍历 for / 增强 for-each、取长度 arr.length
  • 默写：输入 5 个数存数组，找出最大值和下标
  • 练习：Arrays.sort(arr) 排序；数组反转（首尾交换）
  • 坑：下标从 0 开始，arr[5] 会越界报错（明天学怎么优雅处理）
示例代码（先默写，卡住才看） 
import java.util.Arrays;
import java.util.Scanner;

public class ArrayDemo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int[] arr = new int[5];
        for (int i = 0; i < arr.length; i++) {
            System.out.print("输入第 " + (i + 1) + " 个数：");
            arr[i] = sc.nextInt();
        }

        // 找最大值（下标 + 值）
        int max = arr[0], idx = 0;
        for (int i = 1; i < arr.length; i++) {
            if (arr[i] > max) { max = arr[i]; idx = i; }
        }
        System.out.println("最大值 " + max + "，下标 " + idx);

        // 反转（首尾交换，i 只走一半）
        for (int i = 0; i < arr.length / 2; i++) {
            int t = arr[i];
            arr[i] = arr[arr.length - 1 - i];
            arr[arr.length - 1 - i] = t;
        }
        System.out.println("反转后：" + Arrays.toString(arr));
    }
}

Day 5（8.9）方法：参数 / 返回值 / 重载 / 递归 把代码拆成零件
  • 方法五要素：修饰符 / 返回类型 / 方法名 / 参数列表 / 方法体，void 表示无返回
  • 重载：同名方法不同参数（如 max(int,int) 和 max(double,double)）——面试常考
  • 递归：方法自己调自己，必须有"出口"。默写：阶乘 fact(n)
  • 练习：封装一个 getAverage(int[] arr) 返回平均分
示例代码（先默写，卡住才看） 
public class MethodDemo {
    // 重载：名字相同，参数不同，Java 自动匹配
    static int max(int a, int b) { return a > b ? a : b; }
    static double max(double a, double b) { return a > b ? a : b; }

    // 递归：n! = n * (n-1)!，出口是 0! = 1
    static int fact(int n) {
        if (n <= 1) return 1;        // 出口！没有它就会栈溢出
        return n * fact(n - 1);      // 自己调自己
    }

    public static void main(String[] args) {
        System.out.println(max(3, 5));        // 调 int 版
        System.out.println(max(3.5, 2.1));    // 调 double 版
        System.out.println("5! = " + fact(5));
    }
}

// 封装：字段私有，方法公开
class Student {
    private String name;
    private int score;

    public Student(String name, int score) {    // 构造方法：new 时自动调用
        this.name = name;                        // this = 当前对象
        this.score = score;
    }
    public String getName() { return name; }
    public int getScore() { return score; }
}

// 继承 + 多态
class Animal { void speak() { System.out.println("动物叫"); } }
class Dog extends Animal {
    @Override
    void speak() { System.out.println("汪汪"); }  // 方法重写
}

public class OopDemo {
    public static void main(String[] args) {
        Student s = new Student("张三", 85);
        System.out.println(s.getName() + " " + s.getScore());

        Animal a = new Dog();    // 多态：父类引用，子类对象
        a.speak();               // 输出"汪汪"（运行时看真实类型）
    }
}

Day 8（8.12）异常体系 + try-catch-finally 把报错变成可处理的"对象"
  • 搞懂：异常是 Throwable 派生的对象；Error（系统级，不处理）vs Exception（可处理）
  • 默写：try { } catch (Exception e) { } finally { } 三段式，说出 finally 什么时候一定执行
  • 练习：故意写一个数组越界，用 try-catch 接住，对比 Day 4 说的"越界会报错"
示例代码（先默写，卡住才看） 
// 异常 = 一个可以被捕获的对象
public class Demo {
    public static void main(String[] args) {
        try {
            int[] arr = {1, 2, 3};
            System.out.println(arr[10]);          // 数组越界 → 抛 ArrayIndexOutOfBoundsException
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println("下标越界了：" + e.getMessage());
        } catch (Exception e) {                    // 兜底，顺序：子类在前，父类在后
            System.out.println("其他错误：" + e);
        } finally {
            System.out.println("无论成功失败，我都会执行");  // 关资源就放这
        }
    }
}

import java.io.*;

public class CopyFile {

        // 读一个字节数组

}

import java.io.*;

public class ReadLines {
    public static void main(String[] args) throws IOException {
        // 装饰者模式：一层套一层（面试可能问）
        try (BufferedReader br = new BufferedReader(new FileReader("scores.txt"))) {

            while ((line = br.readLine()) != null) {    // 一行一行读，读完返回 null
                System.out.println("读到：" + line);
            }
        }
    }
}

Day 13（8.17）对象序列化 把对象直接存进文件
  • Serializable 接口：对象实现它，就能用 ObjectOutputStream 写进文件（面试常问）
  • serialVersionUID 是什么：版本号，防止类改了读不回来
  • 练习：存一个 Student 对象到 stu.dat，再读出来打印
示例代码（先默写，卡住才看） 
import java.io.*;

class Student implements Serializable {
    private static final long serialVersionUID = 1L;   // 序列化版本号
    String name; int score;
    Student(String n, int s) { name = n; score = s; }
}

public class SerialDemo {
    public static void main(String[] args) throws Exception {
        // 写（序列化）
        try (ObjectOutputStream out = new ObjectOutputStream(new FileOutputStream("stu.dat"))) {
            out.writeObject(new Student("张三", 85));
        }
        // 读（反序列化）
        try (ObjectInputStream in = new ObjectInputStream(new FileInputStream("stu.dat"))) {
            Student s = (Student) in.readObject();
            System.out.println(s.name + " " + s.score);
        }
    }
}

Day 15（8.19）线程创建：Thread + Runnable 程序为什么能"同时"干活
  • 两种创建方式：继承 Thread / 实现 Runnable（推荐后者，理由：Java 单继承）
  • 关键区别：start()（开新线程）vs run()（只是普通方法调用）——面试第一题
  • 默写：两个线程各打印 1-5，观察输出顺序是乱的
示例代码（先默写，卡住才看） 
public class ThreadDemo {
    public static void main(String[] args) {
        // 方式二：实现 Runnable（推荐）
        Runnable task = () -> {                  // Lambda = 简化匿名内部类
            for (int i = 1; i <= 5; i++) {
                System.out.println(Thread.currentThread().getName() + ": " + i);
            }
        };
        Thread t1 = new Thread(task, "线程A");
        Thread t2 = new Thread(task, "线程B");
        t1.start();                              // start() 才是开新线程！
        t2.start();
        // 直接调 t1.run() 只会在主线程顺序执行 —— 自己试试区别
    }
}

Day 16（8.20）线程安全 + synchronized 经典卖票问题
  • 先写"卖票"bug 版：100 张票，3 个线程卖，不加锁 → 票会卖超（负数）
  • 原因：线程切换的瞬间，两个线程同时读了同一个 ticket
  • 解决：synchronized 加锁（锁对象/锁方法），让"查票→减票"变成原子操作
示例代码（先默写，卡住才看） 
class TicketSeller implements Runnable {
    private int tickets = 100;
    @Override
    public synchronized void run() {            // 锁住整个方法：同一时刻只有一个线程进来
        while (tickets > 0) {
            try { Thread.sleep(10); } catch (InterruptedException e) { }
            System.out.println(Thread.currentThread().getName() + " 卖出第 " + tickets-- + " 张票");
        }
    }
}

public class SellDemo {
    public static void main(String[] args) {
        TicketSeller seller = new TicketSeller();
        new Thread(seller, "窗口1").start();
        new Thread(seller, "窗口2").start();
        new Thread(seller, "窗口3").start();
    }
}
✅ 达标自测：先跑不加 synchronized 的版本记录 bug，再加锁跑通，最后能用一句话说清"为什么卖超了"。

Day 17（8.21）线程通信：wait / notify 生产者-消费者
  • 概念：一个线程"生产"、一个线程"消费"，中间用队列缓冲；满了等、空了等
  • wait()（释放锁并等待）/ notify()（叫醒一个）/ notifyAll()（叫醒全部）——必须在 synchronized 块里用
  • 默写：简化版生产者消费者（一个队列，容量 5）
示例代码（先默写，卡住才看） 
import java.util.LinkedList;
import java.util.Queue;

class Box {
    private Queue<Integer> q = new LinkedList<>();
    private int capacity = 5;

    public synchronized void put(int v) throws InterruptedException {
        while (q.size() >= capacity) wait();      // 满了就等，等消费
        q.offer(v);
        System.out.println("生产了 " + v + "，现有 " + q.size());
        notifyAll();                              // 叫醒消费者
    }
    public synchronized int take() throws InterruptedException {
        while (q.isEmpty()) wait();               // 空了就等，等生产
        int v = q.poll();
        notifyAll();
        return v;
    }
}
✅ 达标自测：能说出"为什么 wait 必须在 synchronized 里"（因为要释放锁，没锁没得释放）。

Day 18（8.22）线程池 + Callable 企业里没人裸 new Thread
  • 为什么用线程池：频繁创建/销毁线程太贵 → 复用。面试必背：线程池 7 大参数
  • 用 Executors.newFixedThreadPool(3) 提交 10 个任务，观察只有 3 个线程在干活

示例代码（先默写，卡住才看） 
import java.util.concurrent.*;

public class PoolDemo {
    public static void main(String[] args) throws Exception {
        ExecutorService pool = Executors.newFixedThreadPool(3);  // 固定 3 个线程
        Future<Integer> f = pool.submit(new Callable<Integer>() {
            @Override
            public Integer call() {               // 有返回值！
                return 1 + 1;
            }
        });
        System.out.println("结果 = " + f.get());   // 阻塞等待结果
        pool.shutdown();                          // 用完要关
    }
}

Day 20（8.24）网络编程：Socket 入门 两台程序怎么说话
  • 概念：ServerSocket（服务端，等着被连）vs Socket（客户端，主动连）
  • 默写：客户端连本机 8888 端口，发一句"你好"，服务端收到后回一句"收到"
  • 跑两个控制台窗口（先启动服务端）验证
示例代码（先默写，卡住才看） 
// 服务端：先跑这个
import java.io.*;
import java.net.*;

public class Server {
    public static void main(String[] args) throws IOException {
        ServerSocket ss = new ServerSocket(8888);        // 监听 8888 端口
        System.out.println("等待连接…");
        Socket s = ss.accept();                          // 阻塞，直到有客户端连上来
        BufferedReader in = new BufferedReader(new InputStreamReader(s.getInputStream()));
        System.out.println("收到：" + in.readLine());      // 读客户端发来的一行
        s.getOutputStream().write("收到！\n".getBytes());
        s.close(); ss.close();
    }
}

// 客户端：后跑这个（另一个窗口）
class Client {
    public static void main(String[] args) throws IOException {
        Socket s = new Socket("127.0.0.1", 8888);        // 连本机 8888
        s.getOutputStream().write("你好，服务端\n".getBytes());
        BufferedReader in = new BufferedReader(new InputStreamReader(s.getInputStream()));
        System.out.println("服务端回：" + in.readLine());
        s.close();
    }
}

Day 24（8.28）JDBC 入门 Java 连上 MySQL
  • 下载 MySQL 驱动 jar（mysql-connector-j），在 IDEA 里 Project Structure → Libraries 导入
  • JDBC 六步（面试必背）：加载驱动 → 获取连接 → 创建语句 → 执行 → 处理结果 → 关闭资源
  • 默写：用 PreparedStatement 插入一个学生（为什么用它？防 SQL 注入 + 性能）
示例代码（先默写，卡住才看） 
import java.sql.*;

public class JdbcDemo {
    public static void main(String[] args) throws Exception {
        // 1. 加载驱动（8+ 版本可以省略，但建议写上）
        Class.forName("com.mysql.cj.jdbc.Driver");
        // 2. 获取连接：jdbc:mysql://主机:端口/库名?参数
        String url = "jdbc:mysql://localhost:3306/school_db?useSSL=false&serverTimezone=Asia/Shanghai";
        try (Connection conn = DriverManager.getConnection(url, "root", "你的密码");
             // 3. 预编译语句：? 是占位符
             PreparedStatement ps = conn.prepareStatement(
                 "INSERT INTO student(name, age, score, class_name) VALUES (?, ?, ?, ?)")) {
            // 4. 给占位符赋值
            ps.setString(1, "赵六");
            ps.setInt(2, 22);
            ps.setDouble(3, 77.5);
            ps.setString(4, "计科2303");

            int rows = ps.executeUpdate();
            System.out.println("影响行数：" + rows);
        } // try-with-resources 自动关连接
    }
}

// 需要 druid jar 包 + 项目里建 druid.properties 配置文件
import com.alibaba.druid.pool.DruidDataSourceFactory;
import javax.sql.DataSource;
import java.io.InputStream;
import java.util.Properties;

public class JdbcUtil {
    private static DataSource ds;
    static {
        try {
            Properties p = new Properties();
            InputStream in = JdbcUtil.class.getClassLoader()
                    .getResourceAsStream("druid.properties");
            p.load(in);
            ds = DruidDataSourceFactory.createDataSource(p);  // 从配置建连接池
        } catch (Exception e) { throw new RuntimeException(e); }
    }
    public static Connection getConn() throws Exception {
        return ds.getConnection();              // 从池子里拿（用完还回去）
    }
}
